# Quick Wins Bundle 2 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Ship five small, independent UX fixes: 401 auth handling + a token-aware root route, a
one-click "load my most recent career" action, a two-column Atuações tab with subbed-off player
dimming, and a pre-match team-sheet view.

**Architecture:** One backend task adds a `last_played_at` timestamp touched at the three
existing "the user is playing" call sites (career creation, instant advance, live-match
kickoff) and threads it through the career list endpoint's ordering. Four frontend-only tasks
follow, each touching a small, focused set of files with no cross-task runtime dependency except
Task 3 consuming Task 1's new field.

**Tech Stack:** Java 21 / Spring Boot 4 / Flyway / Postgres Testcontainers (backend); React 19 /
MUI / TanStack Query / Vitest + Testing Library (frontend), pt-BR UI copy.

**Spec:** `docs/superpowers/specs/2026-07-16-quick-wins-2-design.md`

## Global Constraints

- pt-BR UI copy throughout; use the exact strings given in each task.
- 401 handling must NOT fire for `/auth/login` or `/auth/register` — those endpoints return 401
  for bad credentials, which must surface as an inline form error, not a storage wipe + redirect.
- The subbed-off/dimming rule is `used === true or the player is simply absent from the current
  onPitch list` — concretely: **any rated player whose `playerId` is not in the current
  `onPitch` set is dimmed** (this also correctly dims red-carded players, which is intentional).
- No backend changes for subbed-off styling — it's derivable from `LiveMatchState.onPitch` /
  `opponentOnPitch`, both already present.
- `LiveMatchService.advance` (in-match minute polling) must NOT touch `last_played_at` — only
  `createCareer`, `SeasonAdvanceService.advance`, and `LiveMatchService.kickoff` do.
- Backend tests: `cd backend && JAVA_HOME=/Library/Java/JavaVirtualMachines/amazon-corretto-25.jdk/Contents/Home ./gradlew test --tests "<pattern>"`. Frontend: `cd frontend && npx vitest run <file>` and `npx tsc --noEmit`.

Replace `frontend/src/api/client.ts` in full:

```ts
import { clearAuthStorage } from '../auth/authStorage'

const BASE = import.meta.env.VITE_API_URL ?? 'http://localhost:8088/api/v1'

/** An HTTP failure, preserving the backend's error code so callers can react to it precisely. */
export class ApiError extends Error {
  readonly code: string | null
  readonly status: number

  constructor(message: string, code: string | null, status: number) {
    super(message)
    this.name = 'ApiError'
    this.code = code
    this.status = status
  }
}

async function handle<T>(res: Response, path: string): Promise<T> {
  if (!res.ok) {
    const body = (await res.json().catch(() => ({}))) as { error?: string; message?: string }
    // A 401 means the stored token is missing/expired/invalid — except on the auth endpoints
    // themselves, where a 401 just means "wrong username/password" and must surface as an
    // inline form error, not wipe storage and bounce the user away from the login page.
    if (res.status === 401 && !path.startsWith('/auth/')) {
      clearAuthStorage()
      window.location.href = '/login'
    }
    throw new ApiError(body.message ?? res.statusText ?? 'Request failed', body.error ?? null, res.status)
  }
  return res.status === 204 ? (undefined as T) : (res.json() as Promise<T>)
}

export async function apiGet<T>(path: string, token?: string): Promise<T> {
  const res = await fetch(`${BASE}${path}`, {
    headers: token ? { Authorization: `Bearer ${token}` } : {},
  })
  return handle<T>(res, path)
}

export async function apiPost<T>(path: string, body: unknown, token?: string): Promise<T> {
  const res = await fetch(`${BASE}${path}`, {
    method: 'POST',
    headers: {
      'Content-Type': 'application/json',
      ...(token ? { Authorization: `Bearer ${token}` } : {}),
    },
    body: JSON.stringify(body),
  })
  return handle<T>(res, path)
}

export async function apiPut<T>(path: string, body: unknown, token?: string): Promise<T> {
  const res = await fetch(`${BASE}${path}`, {
    method: 'PUT',
    headers: {
      'Content-Type': 'application/json',
      ...(token ? { Authorization: `Bearer ${token}` } : {}),
    },
    body: JSON.stringify(body),
  })
  return handle<T>(res, path)
}

export async function apiDelete<T>(path: string, token?: string): Promise<T> {
  const res = await fetch(`${BASE}${path}`, {
    method: 'DELETE',
    headers: token ? { Authorization: `Bearer ${token}` } : {},
  })
  return handle<T>(res, path)
}
```

import type { QueryClient } from '@tanstack/react-query'

export function DashboardPage() {
  const { user, token } = useAuth()
  const navigate = useNavigate()
  const queryClient = useQueryClient()

  const saveGameKey = saveGameKeyFor(user?.userId)
  const [saveGameId, setSaveGameId] = useState<string | null>(() =>
    localStorage.getItem(saveGameKey),
  )

    queryKey: ['saveGame', saveGameId],

            superior direito e escolha <Box component="strong">Nova Carreira</Box> para começar.
          </Typography>
          {careers && careers.length > 0 && (
            <Button variant="contained" sx={{ mt: 2 }} onClick={loadLatest}>
              Carregar última carreira
            </Button>
          )}
        </>

      {loadingGame && saveGameId && <Typography sx={{ mt: 2 }}>Carregando carreira…</Typography>}

import { Box, Stack, Typography } from '@mui/material'
import type { PlayerRating } from '../../api/types'

export const POSITION_LABEL: Record<string, string> = { GK: 'GOL', DEF: 'DEF', MID: 'MEI', ATT: 'ATA' }

/** Colour band for a 0–10 grade: green (great) → amber (average) → red (poor). */
export function ratingColor(rating: number): string {
  if (rating >= 7.5) return '#1b7e3c'
  if (rating >= 6.5) return '#3e8e41'
  if (rating >= 5.5) return '#c9911f'
  return '#c0392b'
}

/**
 * Presentational list of players' 0–10 match grades, best first. Shared by the live full-time
 * screen and the persisted post-match news summary so both render identically. When
 * `onPitchIds` is provided, any rated player NOT in that set (subbed off, or sent off) renders
 * dimmed — omit it (as the persisted news card does) to render every row at full opacity.
 */
export function PlayerRatingsList({ ratings, onPitchIds }: { ratings: PlayerRating[]; onPitchIds?: Set<string> }) {
  const sorted = [...ratings].sort((a, b) => b.rating - a.rating)
  return (
    <Stack spacing={0.5} sx={{ mt: 0.5 }}>
      {sorted.map(r => {
        const subbedOff = onPitchIds !== undefined && !onPitchIds.has(r.playerId)
        return (
          <Stack
            key={r.playerId}
            direction="row"
            spacing={1}
            sx={{ alignItems: 'center' }}
            style={subbedOff ? { opacity: 0.5 } : undefined}
          >
            <Box
              sx={{
                minWidth: 36, textAlign: 'center', fontWeight: 700, color: '#fff',
                bgcolor: ratingColor(r.rating), borderRadius: 1, px: 0.5, py: '2px', fontSize: '0.85rem',
              }}
            >
              {r.rating.toFixed(1)}
            </Box>
            <Typography sx={{ minWidth: 30, color: 'text.secondary', fontSize: '0.8rem' }}>
              {POSITION_LABEL[r.position] ?? r.position}
            </Typography>
            <Typography sx={{ flex: 1, minWidth: 0 }} noWrap>{r.name}</Typography>
            {subbedOff && (
              <Typography component="span" title="Substituído" aria-label="Substituído" sx={{ color: 'text.secondary' }}>
                ↩
              </Typography>
            )}
            {r.goals > 0 && (
              <Typography component="span" aria-label={`${r.goals} gol(s)`}>
                {'⚽'.repeat(Math.min(r.goals, 3))}
              </Typography>
            )}
            <Typography sx={{ color: 'text.secondary', fontSize: '0.8rem' }}>{r.minutesPlayed}'</Typography>
          </Stack>
        )
      })}
    </Stack>
  )
}
